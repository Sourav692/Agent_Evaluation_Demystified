# 1.6.2, not 1.6.1: langchain-openai==1.6.2 requires langchain-core>=1.6.2.

langchain-core==1.6.2

# LOAD-BEARING. ragas 0.4.3 has an unconditional top-level
#   `from langchain_community.chat_models.vertexai import ChatVertexAI`
# and declares langchain-community with no upper bound, so an unpinned resolve
# import-errors on `import ragas`.
# 0.4.2 deleted that submodule. 0.4.1 still ships a working copy of it and
# `import ragas` succeeds there, so 0.4.1 is a valid pin too — 0.3.31 is the
# conservative choice, not the only one. What must NOT happen is an unpinned
# resolve, which takes 0.4.2+ and breaks `import ragas` outright.
langchain-community==0.3.31
# Pinned, not floated. Resolving ragas or mlflow can pull langgraph back to
# 1.0.x, which drops langgraph.runtime.ExecutionInfo and breaks the graph in
# notebook 02. If a resolver downgrades it, re-pin it.
langgraph==1.2.11
# 2.54.0, not 3.x: ragas 0.4.3 depends on instructor, which caps openai<3.0.0.
